#!/bin/sh
#
# Purpose:
#       fetch MPD_downloads/straininfo.csv
#
# full url:
# 	https://phenome.jax.org/phenomedoc?name=MPD_downloads/straininfo.csv
#
# products:
#       qcreports_db
#       reports_db
#
# HISTORY
#
# 10/26/2026
#	- per Matt Dunn : 
#	Due to high pressure from data mining traffic, 
#	we are currently blocking automated code/script-based requests in phenome. 
#	This is a temporary fix while we work on more long-term solutions.
#	
#	However, the blocking is based on blocking a specific set of user-agents, 
#	so you should be able to add this user-agent to your wget calls to get the request through:
#	--user-agent="MGI-straininfo/1.0 (bhmgiapp01)"
#
# 12/02/2024 lec
#       - wts2-1585/e4g-88/Jax Pheno Strain Info Curl Issue
#
comment="fetch MPD_downloads/straininfo.csv"
REMOTE_SITE=phenome.jax.org
REMOTE_SITE_URL=https://$REMOTE_SITE
REMOTE_DIR=/
LOCAL_DIR=$REMOTE_SITE
REMOTE_FILES="phenomedoc?name=MPD_downloads/straininfo.csv"
LOCAL_FILES="straininfo.csv"
WGET_OPTIONS="-O $LOCAL_FILES --no-parent --no-check-certificate -S -t 10 -nd -m  --user-agent='MGI-straininfo/1.0'"
local_dir_check=true
do_deletes=false
recursive=false
